{{- if .Values.opal.networkPolicy.enabled }}
NOTE: You are using an external database with enabled NetworkPolicy. Make sure to add the connection details to opal.networkPolicy.extraEgress.
{{- end }}

{{- if and .Values.opal.ingress.enabled .Values.opal.networkPolicy.enabled (not .Values.opal.networkPolicy.extraIngress) }}
NOTE: The Opal NetworkPolicy only allows connections from the Helm test pod. Allow your ingress controller via opal.networkPolicy.extraIngress, otherwise requests through the Ingress time out.

1. Get the application URL by running these commands:
{{- if .Values.opal.ingress.enabled }}
{{- range $host := .Values.opal.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.opal.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.opal.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[?(@.name=='http')].nodePort}" services {{ $serviceName }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.opal.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ $serviceName }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ $serviceName }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.opal.service.port }}
{{- else if contains "ClusterIP" .Values.opal.service.type }}
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward service/{{ $serviceName }} 8080:{{ .Values.opal.service.port }}
{{- end }}

2. Get the password of the "administrator" user:
{{- with .Values.opal.auth.administrator.existingSecret }}
  kubectl get secret --namespace {{ $.Release.Namespace }} {{ include "opal.credentials.secretKeyRef" (dict "existingSecret" . "key" "opal-administrator-password" "context" $) | fromYaml | dig "name" "" }} -o jsonpath="{.data.{{ include "opal.credentials.secretKeyRef" (dict "existingSecret" . "key" "opal-administrator-password" "context" $) | fromYaml | dig "key" "" }}}" | base64 -d
{{- end }}
