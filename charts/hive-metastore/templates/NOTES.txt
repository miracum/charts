The Hive Metastore is reachable inside the cluster via its Thrift endpoint:

  thrift://{{ include "hive-metastore.fullname" . }}.{{ .Release.Namespace }}.svc:{{ .Values.service.port }}
{{- if contains "NodePort" .Values.service.type }}

To reach it from outside the cluster via the NodePort:

  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "hive-metastore.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo thrift://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}

NOTE: It may take a few minutes for the LoadBalancer IP to be available.
      You can watch its status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "hive-metastore.fullname" . }}'

  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "hive-metastore.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo thrift://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}

To reach it from your machine, use port-forwarding:

  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "hive-metastore.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}
{{- end }}
