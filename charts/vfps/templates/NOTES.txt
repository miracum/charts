{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "vfps.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status of by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "vfps.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "vfps.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else if contains "ClusterIP" .Values.service.type }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "vfps.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

{{- if lt $replicas 2 }}

NOTE: vfps is running with a single replica, so every rolling upgrade, node drain and pod
      eviction is a gap in service. For a highly-available deployment set `replicaCount` to 2 or
      more, enable `podDisruptionBudget`, and spread the pods with `topologySpreadConstraints`.
{{- end }}

{{- if and (ge $replicas 2) (eq .Values.service.sessionAffinity "None") }}

NOTE: More than one replica is running but no session affinity is configured. The admin UI is
      Blazor Server - a browser session is a SignalR circuit held in one replica's memory and
      cannot be resumed on another - so the UI needs affinity to work reliably. Either add
      cookie-based affinity annotations to your ingress controller (preferred), or set
      `service.sessionAffinity=ClientIP`. The gRPC and REST APIs are unaffected.
{{- end }}

{{- if and .Values.gracefulShutdown.preStopSleepSeconds (not (semverCompare ">=1.30-0" .Capabilities.KubeVersion.Version)) }}

WARNING: `gracefulShutdown.preStopSleepSeconds` is set but this cluster runs Kubernetes
         {{ .Capabilities.KubeVersion.Version }}, which does not support the native `sleep` preStop
         handler (added in 1.30). The hook has been SKIPPED, so requests may still be routed to
         pods that have begun shutting down during a rollout. The vfps image is chiseled and has
         no shell, so an `exec`-based hook is not an alternative - upgrade the cluster, or accept
         the dropped requests.
{{- end }}

{{- if .Values.worker.enabled }}
{{- if lt (int .Values.worker.replicaCount) 1 }}

WARNING: worker.enabled is set but worker.replicaCount is {{ .Values.worker.replicaCount }}. The
         API pods have been switched to accept-only, so with no worker replicas running, nothing
         processes CSV jobs at all - they will queue indefinitely with no error.
{{- else }}

NOTE: CSV pseudonymization jobs run in the separate "{{ include "vfps.fullname" . }}-worker"
      Deployment; the API pods accept and display jobs but no longer process them. This only does
      anything if CSV jobs are enabled in the first place (S3__IsEnabled=true via extraEnv) - the
      chart cannot check that for you. Watch the Hangfire dashboard's Servers page to confirm the
      workers have registered.
{{- end }}
